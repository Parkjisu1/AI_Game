// Reproducible arithmetic for the PM brief. Not a forecast or a live analytics client.
export function wilson(k,n,z=1.959963984540054){
 if(!Number.isInteger(n)||n<=0||!Number.isInteger(k)||k<0||k>n)throw new Error('Expected integers 0 <= retained <= users, users > 0');
 const p=k/n,d=1+z*z/n,c=(p+z*z/(2*n))/d,h=z*Math.sqrt(p*(1-p)/n+z*z/(4*n*n))/d;
 return {rate:p,low:c-h,high:c+h};
}
export function twoArmSample(p0,p1){
 const p=(p0+p1)/2;
 return Math.ceil((1.959963984540054*Math.sqrt(2*p*(1-p))+.8416212335729143*Math.sqrt(p0*(1-p0)+p1*(1-p1)))**2/(p1-p0)**2);
}
